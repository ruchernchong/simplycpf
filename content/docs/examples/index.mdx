## TypeScript contribution calculation

```typescript
type Citizenship =
  | "citizen"
  | "spr-year1"
  | "spr-year2"
  | "spr-year3-plus";

async function calculateCpfContribution(
  contributionMonth: string,
  ordinaryWages: number,
  birthMonth: string,
  citizenship: Citizenship,
) {
  const response = await fetch("https://simplycpf.com/api/cpf/calculate", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      contributionMonth,
      ordinaryWages,
      birthMonth,
      citizenship,
    }),
  });

  const data = await response.json();
  if (!response.ok) throw new Error(data.error);
  return data;
}

const result = await calculateCpfContribution(
  "2026-08",
  5000,
  "1996-04",
  "citizen",
);

console.log(result.contribution);
console.log(result.wageBand);
console.log(result.policy);
```

## Additional Wages with annual context

```typescript
const response = await fetch("https://simplycpf.com/api/cpf/calculate", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({
    contributionMonth: "2026-12",
    ordinaryWages: 8000,
    additionalWages: 20000,
    additionalWageCeilingContext: {
      annualOrdinaryWagesSubjectToCpf: 96000,
      priorAdditionalWagesSubjectToCpf: 0,
    },
    citizenship: "citizen",
    age: 40,
  }),
});
```

The service returns `422` if positive AW is supplied without both annual context values.

## Python monthly projection

```python
import requests

response = requests.post(
    "https://simplycpf.com/api/cpf/projection",
    json={
        "monthlyIncome": 6000,
        "birthDate": "04/1990",
        "startMonth": "2026-08",
        "initialBalances": {
            "oa": 90000,
            "sa": 70000,
            "ma": 50000,
            "ra": 0,
        },
        "endAge": 65,
        "citizenship": "citizen",
        "retirementTransfer": {
            "amount": 200,
            "timing": "monthly",
        },
    },
)
response.raise_for_status()
projection = response.json()

for row in projection["yearlyBalances"]:
    print(row["year"], row["balances"], row["policy"]["status"])

assert projection["cpfLifeEstimate"] is None
print(projection["cpfLifeReference"]["rows"])
```

The returned salary and future-policy assumptions are part of the response. CPF LIFE rows are exact CPF Board references, not personalised estimates.

## Official quarterly interest observations

```typescript
const response = await fetch(
  "https://simplycpf.com/api/cpf/interest-rates/trend",
);
const { observations, methodology, policy } = await response.json();

for (const observation of observations) {
  console.log(
    observation.quarter,
    observation.oaRate,
    observation.saRate,
    observation.maRate,
    observation.raRate,
    observation.sourceUrl,
  );
}

console.log(methodology.specialMediSaveRetirementAccounts);
console.log(policy.sources);
```

This endpoint contains official quarterly declarations. It does not expose a reconstructed monthly SGS series.

## Editable investment assumptions

```python
response = requests.post(
    "https://simplycpf.com/api/cpf/investment-comparison",
    json={
        "principal": 100000,
        "years": 20,
        "scenarios": [
            {"name": "CPF OA floor preset", "rate": 2.5},
            {"name": "User investment assumption", "rate": 6.0},
        ],
    },
)
response.raise_for_status()
print(response.json()["results"])
```

The CPF floor preset is an official policy input. The 6% non-CPF return is a user assumption, not a CPF Board fact or market forecast.
